    22	object Semantics {
    23	
    24	  private def compose(first: Semantics, second: Semantics): Semantics = new Semantics {
    25	
    26	    override def initialize(cpg: Cpg): Unit = {
    27	      second.initialize(cpg)
    28	      first.initialize(cpg)
    29	    }
    30	
    31	    override def forMethod(method: Method): Option[FlowSemantic] =
    32	      second.forMethod(method).orElse { first.forMethod(method) }
    33	  }
    34	}

   101	      },
   102	      regex
   103	    )

    11	object FullNameSemantics {
    12	
    13	  private val logger = LoggerFactory.getLogger(getClass)
    14	
    15	  /** Builds FullNameSemantics given their constituent FlowSemantics. Same methodFullNamed FlowSemantic elements are
    16	    * combined into a single one with both of their FlowMappings.
    17	    */
    18	  def fromList(elements: List[FlowSemantic]): FullNameSemantics = FullNameSemantics(
    19	    elements.groupBy(_.methodFullName).map { (fullName, semantics) =>
    20	      val howMany = semantics.length
    21	      if (howMany > 1) {
    22	        logger.warn(s"$howMany competing FlowSemantics found for $fullName, merging them")
    23	      }
    24	      fullName -> FlowSemantic(
    25	        methodFullName = fullName,
    26	        mappings = semantics.flatMap(_.mappings),
    27	        regex = semantics.exists(_.regex)
    28	      )
    29	    }
    30	  )
    31	
    32	  def empty: FullNameSemantics = fromList(List())
    33	
    34	}
    35	
    36	class FullNameSemantics private (methodToSemantic: Map[String, FlowSemantic]) extends Semantics {
    37	
    38	  /** The map below keeps a mapping between results of a regex and the regex string it matches. e.g.
    39	    *
    40	    * `path/to/file.py:<module>.Foo.sink` -> `^path.*Foo\\.sink$`
    41	    */
    42	  private val regexMatchedFullNames = mutable.HashMap.empty[String, String]
    43	
    44	  /** Initialize all the method semantics that use regex with all their regex results before query time.
    45	    */
    46	  override def initialize(cpg: Cpg): Unit = {
    47	    import io.shiftleft.semanticcpg.language._
    48	
    49	    methodToSemantic.filter(_._2.regex).foreach { case (regexString, _) =>
    50	      cpg.method.fullName(regexString).fullName.foreach { methodMatch =>
    51	        regexMatchedFullNames.put(methodMatch, regexString)
    52	      }
    53	    }
    54	  }
    55	
    56	  def elements: List[FlowSemantic] = methodToSemantic.values.toList
    57	
    58	  private def forMethod(fullName: String): Option[FlowSemantic] = regexMatchedFullNames.get(fullName) match {
    59	    case Some(matchedFullName) => methodToSemantic.get(matchedFullName)
    60	    case None                  => methodToSemantic.get(fullName)
    61	  }
    62	
    63	  override def forMethod(method: Method): Option[FlowSemantic] = forMethod(method.fullName)
    64	
    65	  def serialize: String = {
    66	    elements
    67	      .sortBy(_.methodFullName)
    68	      .map { elem =>
    69	        s"\"${elem.methodFullName}\" " + elem.mappings
    70	          .collect { case FlowMapping(x, y) => s"$x -> $y" }
    71	          .mkString(" ")
    72	      }
    73	      .mkString("\n")
    74	  }
    75	
    76	  /** Immutably extends the current `FullNameSemantics` with `extraFlows`.
    77	    */
    78	  def plus(extraFlows: List[FlowSemantic]): FullNameSemantics = FullNameSemantics.fromList(elements ++ extraFlows)
    79	
    80	}

   218	  /** Resolves the path to the swift-demangle command and combines it with the provided arguments. This method

   234	        findInStdOut(WhichSwiftCompilerCommand)
   235	          .map { outLine =>
   236	            val resolvedPath = Paths.get(outLine).toRealPath().resolveSibling("swift-demangle").toString
   237	            resolvedPath +: args
   238	          }
   239	      case io.joern.x2cpg.utils.Environment.OperatingSystemType.Mac =>
   240	        // On macOS, we have to ask xcrun --find swift-demangle
   241	        findInStdOut(XcrunFindDemangleCommand).map { outLine => outLine +: args }
   242	      case io.joern.x2cpg.utils.Environment.OperatingSystemType.Unknown =>
   243	        Some(defaultCommand)
   244	    }
   245	    osSpecificCommand.getOrElse(defaultCommand)
   246	  }

   248	  /** Returns the command to get the version of swift-demangle.
   249	    *
   250	    * This method constructs a command sequence that includes the path to the swift-demangle executable (resolved using
   251	    * `resolveSwiftDemangleCommand`) along with the --version flag.
   252	    *
   253	    * @return
   254	    *   A sequence containing the swift-demangle command with the --version argument
   255	    */
   256	  private def swiftDemangleVersionCommand(): Seq[String] = {
   257	    resolveSwiftDemangleCommand(Seq("--version"))
   258	  }

   261	    *
   262	    * This method constructs a command sequence that includes the path to the swift-demangle executable (resolved using
   263	    * `resolveSwiftDemangleCommand`) along with the following options:
   264	    *   - `--compact`: Produces more compact demangled output
   265	    *   - `--no-sugar`: Avoids syntactic sugar in the demangled output, showing raw type information
   266	    *
   267	    * @return
   268	    *   A sequence containing the swift-demangle command with formatting options
   269	    */
   270	  private def swiftDemangleCommand(): Seq[String] = {
   271	    resolveSwiftDemangleCommand(Seq("--compact", "--no-sugar"))
   272	  }

   274	  /** Checks if the current environment has all the required tools to perform Swift type extraction.
   275	    *
   276	    * This method verifies:
   277	    *   - If Package.swift exists (when buildLogPath is not provided)
   278	    *   - If Swift/swiftc are available with version >= 6.1
   279	    *   - If swift-demangle tool is available
   280	    *
   281	    * @param config
   282	    *   The Swift CPG generation configuration
   283	    * @return
   284	    *   true if the environment has all required tools with compatible versions, false otherwise
   285	    */
   286	  private def isValidEnvironment(config: Config): Boolean = {
   287	    if (config.buildLogPath.isEmpty && !Paths.get(config.inputPath, "Package.swift").toFile.canRead) {
   288	      logger.warn("Package.swift not found or can not be read. This does not look like a valid SwiftPM project.")
   289	      return false
   290	    }
   291	
   292	    val commands = if (config.buildLogPath.isDefined) {
   293	      // we do not need 'swift' on the system if the commands are taken from a build log file
   294	      Seq(SwiftCompilerVersionCommand)
   295	    } else {
   296	      SwiftVersionCommands
   297	    }
   298	    val hasSwift = commands.forall { command =>
   299	      findInStdOut(command, _.contains("Swift version ")) match {
   300	        case Some(outLine) =>
   301	          val swiftVersion = outLine.substring(outLine.indexOf("Swift version ") + 14, outLine.indexOf(" ("))
   302	          val isCompatible = Try(VersionHelper.compare(swiftVersion, MinimumSwiftVersion)).toOption.getOrElse(-1) >= 0
   303	          if (!isCompatible) {
   304	            logger.warn(s"Found Swift version '$swiftVersion' but '$MinimumSwiftVersion' is required!")
   305	          }
   306	          isCompatible
   307	        case None =>
   308	          logger.warn("Unable to determine a Swift version on this system!")
   309	          false
   310	      }
