# Deterministic catalog content manifest generation.
# Run from the repository root after all other catalog evidence files are present.
python3 - <<'PY'
import hashlib
import json
from pathlib import Path

root = Path('evidence/swift-v2-qualification-220/joern-catalog')
files = {
    str(path.relative_to(root)): hashlib.sha256(path.read_bytes()).hexdigest()
    for path in sorted(root.rglob('*'))
    if path.is_file() and path.name not in {'content-manifest.json', 'content-manifest.sha256'}
}
manifest = (json.dumps(files, indent=2, sort_keys=True) + '\n').encode()
(root / 'content-manifest.json').write_bytes(manifest)
digest = hashlib.sha256(manifest).hexdigest()
(root / 'content-manifest.sha256').write_text(f'{digest}  content-manifest.json\n')
PY
